// Writes a wire bent into a heart: a circle of radius `r` swept along the heart curve
//   x(t) = W/32 · 16 sin t (sin²t + 0.6)/1.6,
//   z(t) = W/32 · (13 cos t − 5 cos 2t − 2 cos 3t − cos 4t),
// a smooth heart W wide (the classic curve's x = 16 sin³t eased so its tip and dimple are
// rounded: the wire bends no tighter than 0.88 W/32), in the plane y = 0, for t from π + d to
// 3π − d: open at the tip (t = π) by a feed gap W · 0.02621 wide. One body, `wire`.
//
// Run outside the EMCraft workspace, in a crate depending on solidcraft-doc, solidcore-record,
// serde_json and solidcraft-exchange at the commit EMCraft pins (Cargo.toml):
//   cargo run --bin heart -- crates/templates/designs/heart-wire.solidcraft
use solidcore_record::{Vec2, Vec3};
use solidcraft_doc::sketch::{ConstraintKind, EquationCurve, PointExprs, Sketch, VariableUnit};
use solidcraft_doc::{Document, Expr, FeatureKind, PlaneRef};

const DEFAULTS: [(&str, f64, &str); 2] = [("W", 38.0, "mm"), ("r", 0.5, "mm")];
/// Half the gap's angle at the tip.
const D: f64 = 0.07;
/// Points along the path.
const POINTS: usize = 60;

fn value(name: &str) -> f64 {
    DEFAULTS.iter().find(|(n, _, _)| *n == name).map(|(_, v, _)| *v).unwrap()
}

fn heart(w: f64, t: f64) -> (f64, f64) {
    let s = w / 32.0;
    let x = s * 16.0 * t.sin() * (t.sin() * t.sin() + 0.6) / 1.6;
    let z = s * (13.0 * t.cos() - 5.0 * (2.0 * t).cos() - 2.0 * (3.0 * t).cos() - (4.0 * t).cos());
    (x, z)
}

fn main() {
    let args: Vec<String> = std::env::args().collect();
    let mut doc = Document::new("Heart wire");
    for (name, v, unit) in DEFAULTS {
        doc.set_param(name, &v.to_string(), Some(unit), None).unwrap();
    }
    let w = value("W");
    let (t0, t1) = (std::f64::consts::PI + D, 3.0 * std::f64::consts::PI - D);
    // The path in the plane y = 0 (sketch x along x, sketch y along z).
    let mut path = Sketch::new();
    let at: Vec<Vec2> = (0..=POINTS)
        .map(|k| {
            let t = t0 + (t1 - t0) * k as f64 / POINTS as f64;
            let (x, z) = heart(w, t);
            Vec2 { x, y: z }
        })
        .collect();
    let equation = EquationCurve {
        curve: String::new(),
        x: Some("W / 32 * 16 * sin(t) * (sin(t) * sin(t) + 0.6) / 1.6".into()),
        y: "W / 32 * (13 * cos(t) - 5 * cos(2 * t) - 2 * cos(3 * t) - cos(4 * t))".into(),
        start: format!("pi + {D}"),
        end: format!("3 * pi - {D}"),
        tolerance: 1e-4,
        variable_unit: VariableUnit::Number,
    };
    let curve = path.add_equation_curve(&at, false, equation).unwrap();
    let curve_id = path.curves[curve].id.clone();
    let plane = PlaneRef::Placed { origin: [Expr::new("0"), Expr::new("0"), Expr::new("0")], x_dir: Vec3::X, y_dir: Vec3::Z };
    let path_id = doc.add_feature(FeatureKind::Sketch { plane, sketch: path }, Some("heart")).unwrap();
    // The wire's section at the path's start, square to the path there (its direction fixed by
    // D alone, so the same for every W).
    let (h, (xa, za), (xb, zb)) = (1e-6, heart(1.0, t0 - 1e-6), heart(1.0, t0 + 1e-6));
    let _ = h;
    let (tx, tz) = (xb - xa, zb - za);
    let n = (tx * tx + tz * tz).sqrt();
    let (tx, tz) = (tx / n, tz / n);
    // In-plane normal to the tangent (in xz) and y: the section's axes.
    let x_dir = Vec3::Y;
    let y_dir = Vec3 { x: -tz, y: 0.0, z: tx };
    let mut sk = Sketch::new();
    let c = sk.add_point(Vec2 { x: 0.0, y: 0.0 }, None).unwrap();
    sk.points[c].exprs = Some(Box::new(PointExprs { x: Some("0".into()), y: Some("0".into()) }));
    sk.points[c].fixed = true;
    let circle = sk.add_circle(Vec2 { x: 0.0, y: 0.0 }, value("r"), Some(c), None).unwrap();
    sk.add_constraint(ConstraintKind::Radius { c: circle, value: value("r") }, Some("r".into())).unwrap();
    // Feature expressions read a plain number in degrees: the angle given in radians.
    let a = format!("((pi + {D}) * rad)");
    let x0 = format!("W / 32 * 16 * sin({a}) * (sin({a}) * sin({a}) + 0.6) / 1.6");
    let z0 = format!("W / 32 * (13 * cos({a}) - 5 * cos(2 * {a}) - 2 * cos(3 * {a}) - cos(4 * {a}))");
    let plane = PlaneRef::Placed { origin: [Expr::new(&x0), Expr::new("0"), Expr::new(&z0)], x_dir, y_dir };
    let section = doc.add_feature(FeatureKind::Sketch { plane, sketch: sk }, Some("section")).unwrap();
    let mut json: serde_json::Value = serde_json::from_str(&doc.to_json()).unwrap();
    let next = json["next_id"].as_u64().unwrap();
    json["features"].as_array_mut().unwrap().push(serde_json::json!({
        "id": next, "name": "wire", "suppressed": false, "body_names": ["wire"], "type": "sweep",
        "sketch": section, "profiles": {"type": "all"}, "path_sketch": path_id, "path": [curve_id],
        "operation": "new_body"
    }));
    json["next_id"] = serde_json::json!(next + 1);
    let text = serde_json::to_string_pretty(&json).unwrap();
    std::fs::write(&args[1], &text).unwrap();
    let design = solidcraft_exchange::open(text.as_bytes()).unwrap();
    let snap = design.snapshot(&Default::default()).unwrap();
    for b in &snap.bodies {
        let mut v = 0.0;
        for f in &b.faces {
            for tri in &f.triangles {
                let p = tri.map(|k| b.positions[k.0 as usize].0);
                let (a, c, d) = (p[0], p[1], p[2]);
                v += (a[0] * (c[1] * d[2] - c[2] * d[1]) - a[1] * (c[0] * d[2] - c[2] * d[0]) + a[2] * (c[0] * d[1] - c[1] * d[0])) / 6.0;
            }
        }
        println!("{} faces {} triangles {} closed {} volume {v}", b.name, b.faces.len(), b.triangle_count(), b.closed);
        let mut lo = [f64::MAX; 3];
        let mut hi = [f64::MIN; 3];
        for q in &b.positions {
            for k in 0..3 {
                lo[k] = lo[k].min(q.0[k]);
                hi[k] = hi[k].max(q.0[k]);
            }
        }
        println!("bounds {lo:?} {hi:?}");
        for f in &b.faces {
            let mut fv = 0.0;
            for tri in &f.triangles {
                let p = tri.map(|k| b.positions[k.0 as usize].0);
                let (a, c, d) = (p[0], p[1], p[2]);
                fv += (a[0] * (c[1] * d[2] - c[2] * d[1]) - a[1] * (c[0] * d[2] - c[2] * d[0]) + a[2] * (c[0] * d[1] - c[1] * d[0])) / 6.0;
            }
            println!("face {} tris {} vol {fv}", f.name, f.triangles.len());
        }
    }
    println!("tangent at start ({tx:.6}, {tz:.6}); gap {:.5} W", 2.0 * heart(1.0, std::f64::consts::PI - D).0);
}
