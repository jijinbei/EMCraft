// Writes a circular guide whose wall waves sinusoidally about its axis — r = r0 + h sin(2π(z − l)/p)
// for n periods between two straight leads l long — with SolidCraft's document API: the air
// inside it the one body, `guide`, a profile in the z–r half-plane (sketch x along z, sketch y
// along x) revolved about z. The wave is an equation curve, placed again from the formula
// whenever a parameter changes, fitted within 1e-4 mm.
//
// Run outside the EMCraft workspace, in a crate depending on solidcraft-doc, solidcore-record,
// serde_json and solidcraft-exchange at the commit EMCraft pins (Cargo.toml):
//   cargo run --bin circ -- crates/templates/designs/corrugated-cell.solidcraft cell
use solidcore_record::Vec2;
use solidcraft_doc::sketch::{EquationCurve, PointExprs, Sketch, VariableUnit};
use solidcraft_doc::{Document, Expr, FeatureKind, PlaneRef};

const DEFAULTS: [(&str, f64, Option<&str>); 5] =
    [("r0", 12.0, Some("mm")), ("h", 1.0, Some("mm")), ("p", 16.0, Some("mm")), ("n", 8.0, Some("none")), ("l", 20.0, Some("mm"))];

/// Points along each period of the wave.
const PER_PERIOD: usize = 24;

fn value(name: &str) -> f64 {
    DEFAULTS.iter().find(|(n, _, _)| *n == name).map(|(_, v, _)| *v).unwrap()
}

fn corner(sk: &mut Sketch, x: &str, y: &str, at: [f64; 2]) -> usize {
    let i = sk.add_point(Vec2 { x: at[0], y: at[1] }, None).unwrap();
    sk.points[i].exprs = Some(Box::new(PointExprs { x: Some(x.into()), y: Some(y.into()) }));
    sk.points[i].fixed = true;
    i
}

fn main() {
    let args: Vec<String> = std::env::args().collect();
    // `cell`: one period alone, the wave from z = 0 to n p with no leads (a periodic cell).
    let cell = args.get(2).is_some_and(|a| a == "cell");
    let mut doc = Document::new(if cell { "Circular corrugated cell" } else { "Circular corrugated guide" });
    for (name, v, unit) in DEFAULTS.iter().filter(|d| !cell || d.0 != "l") {
        let v = if cell && name == &"n" { 1.0 } else { *v };
        doc.set_param(name, &v.to_string(), *unit, None).unwrap();
    }
    let (r0, h, p, n, l) = (value("r0"), value("h"), value("p"), if cell { 1.0 } else { value("n") }, if cell { 0.0 } else { value("l") });
    let length = 2.0 * l + n * p;
    let mut sk = Sketch::new();
    let total = if cell { "n * p" } else { "2 * l + n * p" };
    let p0 = corner(&mut sk, "0", "0", [0.0, 0.0]);
    let p1 = corner(&mut sk, total, "0", [length, 0.0]);
    let (p2, p3) = if cell { (0, 0) } else { (corner(&mut sk, total, "r0", [length, r0]), corner(&mut sk, "0", "r0", [0.0, r0])) };
    let count = PER_PERIOD * n as usize + 1;
    let at: Vec<Vec2> = (0..count)
        .map(|k| {
            let z = l + n * p * k as f64 / (count - 1) as f64;
            Vec2 { x: z, y: r0 + h * (2.0 * std::f64::consts::PI * (z - l) / p).sin() }
        })
        .collect();
    let equation = EquationCurve {
        curve: String::new(),
        x: None,
        y: if cell { "r0 + h * sin(2 * pi * (x - 0 mm) / p)".into() } else { "r0 + h * sin(2 * pi * (x - l) / p)".into() },
        start: if cell { "0".into() } else { "l".into() },
        end: if cell { "n * p".into() } else { "l + n * p".into() },
        tolerance: 1e-4,
        variable_unit: VariableUnit::Number,
    };
    let wave = sk.add_equation_curve(&at, false, equation).unwrap();
    let pts = match &sk.curves[wave].kind {
        solidcraft_doc::sketch::CurveKind::Spline { pts, .. } => pts.clone(),
        _ => unreachable!(),
    };
    let (first, last) = (pts[0], *pts.last().unwrap());
    let lines = if cell { vec![(p0, p1), (p1, last), (first, p0)] } else { vec![(p0, p1), (p1, p2), (p2, last), (first, p3), (p3, p0)] };
    for (u, v) in lines {
        sk.add_line_pts(u, v, None).unwrap();
    }
    // The sketch's x along the model's z, its y along x: the half-plane φ = 0.
    let plane = PlaneRef::Placed {
        origin: [Expr::new("0"), Expr::new("0"), Expr::new("0")],
        x_dir: solidcore_record::Vec3::Z,
        y_dir: solidcore_record::Vec3::X,
    };
    let id = doc.add_feature(FeatureKind::Sketch { plane, sketch: sk }, Some("profile")).unwrap();
    let mut json: serde_json::Value = serde_json::from_str(&doc.to_json()).unwrap();
    let next = json["next_id"].as_u64().unwrap();
    json["features"].as_array_mut().unwrap().push(serde_json::json!({
        "id": next, "name": "guide", "suppressed": false, "body_names": ["guide"], "type": "revolve",
        "sketch": id, "profiles": {"type": "all"}, "axis": {"type": "sketch_axis", "axis": "x"},
        "angle": "360 deg", "operation": "new_body", "convention": "sketch_normal"
    }));
    json["next_id"] = serde_json::json!(next + 1);
    let text = serde_json::to_string_pretty(&json).unwrap();
    std::fs::write(&args[1], &text).unwrap();
    let design = solidcraft_exchange::open(text.as_bytes()).unwrap();
    let snap = design.snapshot(&Default::default()).unwrap();
    for b in &snap.bodies {
        let kinds: Vec<String> = b
            .faces
            .iter()
            .map(|f| {
                let k = match f.surface {
                    solidcraft_exchange::Surface::Nurbs { .. } => "nurbs",
                    solidcraft_exchange::Surface::Plane { .. } => "plane",
                    solidcraft_exchange::Surface::Cylinder { .. } => "cyl",
                    _ => "other",
                };
                format!("{}:{k}", f.name)
            })
            .collect();
        println!("{} faces {} triangles {} {:?}", b.name, b.faces.len(), b.triangle_count(), kinds);
    }
}
