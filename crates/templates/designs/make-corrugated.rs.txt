// Writes a corrugated guide with SolidCraft's document API: a rectangular guide along z one of
// whose walls waves sinusoidally for n periods between two straight leads l long — the broad
// wall x = a + h sin(2π(z − l)/p) (`h`: an H-plane corrugation) or the narrow wall
// y = b + h sin(2π(z − l)/p) (`e`: an E-plane one); the air inside it is the one body,
// `guide`, a profile in the z–x (or z–y) plane extruded along y (or x). The wave is an equation
// curve (its points placed again from the formula whenever a parameter changes), fitted within
// 1e-4 mm.
//
// Run outside the EMCraft workspace, in a crate depending on solidcraft-doc, solidcore-record,
// serde_json and solidcraft-exchange at the commit EMCraft pins (Cargo.toml):
//   cargo run --bin corrugated -- h crates/templates/designs/corrugated-h.solidcraft
//   cargo run --bin corrugated -- e crates/templates/designs/corrugated-e.solidcraft
use solidcore_record::Vec2;
use solidcraft_doc::sketch::{EquationCurve, PointExprs, Sketch, VariableUnit};
use solidcraft_doc::{Document, Expr, FeatureKind, PlaneRef};

const DEFAULTS: [(&str, f64, Option<&str>); 6] =
    [("a", 22.86, Some("mm")), ("b", 10.16, Some("mm")), ("h", 2.0, Some("mm")), ("p", 10.0, Some("mm")), ("n", 4.0, Some("none")), ("l", 10.0, Some("mm"))];

/// Points along each period of the wave (a cubic fit through 24 points a period strays about
/// 2e-5 h from a sine).
const PER_PERIOD: usize = 24;

fn value(name: &str) -> f64 {
    DEFAULTS.iter().find(|(n, _, _)| *n == name).map(|(_, v, _)| *v).unwrap()
}

/// A fixed point placed by expressions (sketch x along z, sketch y along x).
fn corner(sk: &mut Sketch, x: &str, y: &str, at: [f64; 2]) -> usize {
    let i = sk.add_point(Vec2 { x: at[0], y: at[1] }, None).unwrap();
    sk.points[i].exprs = Some(Box::new(PointExprs { x: Some(x.into()), y: Some(y.into()) }));
    sk.points[i].fixed = true;
    i
}

fn main() {
    let args: Vec<String> = std::env::args().collect();
    let e_plane = args[1] == "e";
    // The wall that waves (its side, the profile's height) and the guide's other side (the
    // extrusion's length); the sketch's y along x (H-plane) or y (E-plane).
    let (side, depth) = if e_plane { ("b", "a") } else { ("a", "b") };
    let mut doc = Document::new(if e_plane { "E-plane corrugated guide" } else { "H-plane corrugated guide" });
    for (name, v, unit) in DEFAULTS {
        doc.set_param(name, &v.to_string(), unit, None).unwrap();
    }
    let (a, h, p, n, l) = (value(side), value("h"), value("p"), value("n"), value("l"));
    let length = 2.0 * l + n * p;
    let mut sk = Sketch::new();
    let p0 = corner(&mut sk, "0", "0", [0.0, 0.0]);
    let p1 = corner(&mut sk, "2 * l + n * p", "0", [length, 0.0]);
    let p2 = corner(&mut sk, "2 * l + n * p", side, [length, a]);
    let p3 = corner(&mut sk, "0", side, [0.0, a]);
    // The wave, x = a + h sin(2π(z − l)/p), from z = l to l + n p.
    let count = PER_PERIOD * n as usize + 1;
    let at: Vec<Vec2> = (0..count)
        .map(|k| {
            let z = l + n * p * k as f64 / (count - 1) as f64;
            Vec2 { x: z, y: a + h * (2.0 * std::f64::consts::PI * (z - l) / p).sin() }
        })
        .collect();
    let equation = EquationCurve {
        curve: String::new(),
        x: None,
        y: format!("{side} + h * sin(2 * pi * (x - l) / p)"),
        start: "l".into(),
        end: "l + n * p".into(),
        tolerance: 1e-4,
        variable_unit: VariableUnit::Number,
    };
    let wave = sk.add_equation_curve(&at, false, equation).unwrap();
    let pts = match &sk.curves[wave].kind {
        solidcraft_doc::sketch::CurveKind::Spline { pts, .. } => pts.clone(),
        _ => unreachable!(),
    };
    let (first, last) = (pts[0], *pts.last().unwrap());
    for (u, v) in [(p0, p1), (p1, p2), (p2, last), (first, p3), (p3, p0)] {
        sk.add_line_pts(u, v, None).unwrap();
    }
    // The sketch's x along the model's z, its y along x (extruded along z × x = y) or along y
    // (extruded against z × y = −x, so along x).
    let plane = PlaneRef::Placed {
        origin: [Expr::new("0"), Expr::new("0"), Expr::new("0")],
        x_dir: solidcore_record::Vec3::Z,
        y_dir: if e_plane { solidcore_record::Vec3::Y } else { solidcore_record::Vec3::X },
    };
    let id = doc.add_feature(FeatureKind::Sketch { plane, sketch: sk }, Some("profile")).unwrap();
    let mut json: serde_json::Value = serde_json::from_str(&doc.to_json()).unwrap();
    let next = json["next_id"].as_u64().unwrap();
    json["features"].as_array_mut().unwrap().push(serde_json::json!({
        "id": next, "name": "guide", "suppressed": false, "body_names": ["guide"], "type": "extrude",
        "sketch": id, "profiles": {"type": "all"},
        "extent": if e_plane { serde_json::json!({"distance": depth, "direction": "negative"}) } else { serde_json::json!({"distance": depth}) },
        "operation": "new_body"
    }));
    json["next_id"] = serde_json::json!(next + 1);
    let text = serde_json::to_string_pretty(&json).unwrap();
    std::fs::write(&args[2], &text).unwrap();
    let design = solidcraft_exchange::open(text.as_bytes()).unwrap();
    let snap = design.snapshot(&Default::default()).unwrap();
    for b in &snap.bodies {
        println!("{} faces {}", b.name, b.faces.len());
    }
}
